<script>
    let statusChartInstance = null;
    let currentIsAdmin = false;

                .file-pdf { border-color: #ef4444; background-color: #fef2f2; color: #991b1b !important; }

    function setEditorMode(mode) {
        const editor = tinymce.get('sharedReportEditor');
        if (!editor) return;

        const btnView = document.getElementById('btnViewMode');
        const btnEdit = document.getElementById('btnEditMode');
        const btnSave = document.getElementById('btnSaveReport');

        if (mode === 'view') {
            editor.mode.set('readonly');
            btnView.classList.add('btn-primary', 'active');
            btnView.classList.remove('btn-outline-secondary');
            btnEdit.classList.add('btn-outline-secondary');
            btnEdit.classList.remove('btn-primary', 'active');
            btnSave.style.display = 'none';
        } else {
            editor.mode.set('design');
            btnEdit.classList.add('btn-primary', 'active');
            btnEdit.classList.remove('btn-outline-secondary');
            btnView.classList.add('btn-outline-secondary');
            btnView.classList.remove('btn-primary', 'active');
            btnSave.style.display = 'inline-block';
        }
    }

    function loadSharedReport() {
        const yearSelect = document.getElementById('yearFilter');
        let selectedYear = yearSelect ? yearSelect.value : new Date().getFullYear().toString();
        
        google.script.run.withSuccessHandler(content => {
            const editor = tinymce.get('sharedReportEditor');
            if (editor) editor.setContent(content || "");
        }).getSharedReport(selectedYear); // ส่งปีที่เลือกไปหลังบ้าน
    }


    function saveSharedReport() {
        const editor = tinymce.get('sharedReportEditor');
        if (!editor) return; 
        
        const yearSelect = document.getElementById('yearFilter');
        let selectedYear = yearSelect ? yearSelect.value : new Date().getFullYear().toString();
        if (selectedYear === "All") {
            selectedYear = new Date().getFullYear().toString(); // ถ้าเลือก All จะบันทึกเป็นปีปัจจุบัน
        }
        
        const content = editor.getContent();
        const btn = document.getElementById('btnSaveReport');
        const originalText = btn.innerHTML;
        
        btn.innerHTML = '<span class="spinner-border spinner-border-sm"></span> Saving...';
        btn.disabled = true;

        google.script.run.withSuccessHandler(() => {
            alert('✅ Report details saved for year ' + selectedYear + ' successfully!');
            btn.innerHTML = originalText;
            btn.disabled = false;
            setEditorMode('view');
        }).saveSharedReport(content, selectedYear); // ส่งปีไปคู่กับข้อมูล
    }


    async function uploadDirectToDoc(input) {
        const files = input.files;
        if (files.length === 0) return;

        const editor = tinymce.activeEditor;
        
        for (let i = 0; i < files.length; i++) {
            let file = files[i];
            let fileId = 'loading-' + Math.random().toString(36).substring(2, 9);
            
            let loadingHtml = `<span id="${fileId}" class="upload-loading-box" contenteditable="false">⏳ กำลังอัปโหลด: ${file.name}...</span>`;
            editor.insertContent(loadingHtml);

            let reader = new FileReader();
            reader.readAsDataURL(file);

            await new Promise((resolve) => {
                reader.onload = () => {
                    google.script.run.withSuccessHandler(url => {
                        if (url) {
                            let ext = file.name.split('.').pop().toLowerCase();
                            let colorClass = ''; let icon = '📄';

                            if (ext === 'pdf') { colorClass = 'file-pdf'; icon = '📕'; } 
                            else if (['xlsx', 'xls', 'csv'].includes(ext)) { colorClass = 'file-excel'; icon = '📗'; } 
                            else if (['docx', 'doc'].includes(ext)) { colorClass = 'file-word'; icon = '📘'; } 
                            else if (['png', 'jpg', 'jpeg', 'gif', 'webp'].includes(ext)) { colorClass = 'file-image'; icon = '🖼️'; }

                            let finalHtml = `<a href="${url}" target="_blank" class="file-link-box ${colorClass}" contenteditable="false">${icon} ${file.name}</a>&nbsp;`;
                            
                            let loadingNode = editor.dom.select(`#${fileId}`)[0];
                            if (loadingNode) editor.dom.setOuterHTML(loadingNode, finalHtml);
                            else editor.insertContent(finalHtml);
                        } else {
                            let loadingNode = editor.dom.select(`#${fileId}`)[0];
                            if (loadingNode) editor.dom.remove(loadingNode);
                            alert('อัปโหลดไฟล์ ' + file.name + ' ผิดพลาด');
                        }
                        resolve();
                    }).uploadFile(reader.result, file.name);
                };
            });
        }
        input.value = ''; 
    }

    function toggleForm() {
        const form = document.getElementById('formSection');
        form.style.display = form.style.display === 'none' ? 'block' : 'none';
        if(form.style.display === 'block') form.scrollIntoView({ behavior: 'smooth' });
    }

    function uploadSingleFile(fileInputId) {
        return new Promise((resolve) => {
            const file = document.getElementById(fileInputId).files[0];
            if (!file) { resolve(""); return; }
            
            const reader = new FileReader();
            reader.readAsDataURL(file);
         
            reader.onload = () => {
                google.script.run
                    .withSuccessHandler(url => resolve(url))
                    .uploadFile(reader.result, file.name);
            };
        });
    }

    async function processForm() {
        const btn = document.getElementById('btnSubmit');
        btn.innerHTML = '<span class="spinner-border spinner-border-sm"></span> SAVING...';
        btn.disabled = true;

        try {
            const selectedFindings = Array.from(document.querySelectorAll('input[name="finding"]:checked')).map(cb => cb.value).join(', ');
            
            const [reportUrl, carUrl, whyUrl, yokotenUrl] = await Promise.all([
                uploadSingleFile('reportFiles'), uploadSingleFile('carFiles'),
                uploadSingleFile('whyFiles'), uploadSingleFile('yokotenFiles')
            ]);
            
            const payload = {
                hiddenAuditId: document.getElementById('hiddenAuditId').value,
                auditDate: document.getElementById('auditDate').value,
                plant: document.getElementById('plant').value,
                dept: document.getElementById('dept').value,
                findings: selectedFindings,             
                detail: document.getElementById('detail').value,
                reportFiles: reportUrl,
                carDetail: document.getElementById('carDetail').value,
                carFiles: carUrl,
                whyDetail: document.getElementById('whyDetail').value,
                whyFiles: whyUrl,
                carEmail: document.getElementById('carEmail').value,
                carDeadline: document.getElementById('carDeadline').value,

                fu3: document.getElementById('fu3').value,

                yokotenFiles: yokotenUrl

                    document.getElementById('auditForm').reset();
                    document.getElementById('hiddenAuditId').value = ""; 
                    document.getElementById('statusCar').disabled = false; 
                    document.getElementById('evStatus').disabled = false;
                    
                    btn.innerHTML = 'SUBMIT DATA'; 
                    btn.classList.replace('btn-warning', 'btn-mnb');

                })

            alert("Error: " + e);

        }
    }

    function formatEnglishDate(dateString) {
        if (!dateString || dateString === "-" || dateString.trim() === "") return dateString;
        let d;
        if (dateString.includes('/')) {
            let parts = dateString.split('/');
            if (parts[0].length <= 2) d = new Date(`${parts[2]}-${parts[1]}-${parts[0]}`);
            else d = new Date(dateString);
        } else d = new Date(dateString);
        if (isNaN(d.getTime())) return dateString;
        return d.toLocaleDateString('en-GB', { day: '2-digit', month: 'short', year: 'numeric' });
    }

    function loadDashboardData() {
        const selectedYear = document.getElementById('yearFilter') ? document.getElementById('yearFilter').value : "All";
        
        // 🌟 สั่งให้ดึง Doc Report ของปีนั้นๆ มาแสดงด้วยทันที
        loadSharedReport(); 
        
        google.script.run.withSuccessHandler(data => {
            currentIsAdmin = data.isAdmin; 
            
            const yearSelect = document.getElementById('yearFilter');
            if (yearSelect && yearSelect.options.length <= 1 && data.availableYears.length > 0) {
                data.availableYears.forEach(y => {
                    let opt = document.createElement('option');
                    opt.value = y; opt.text = "Year " + y;
                    yearSelect.appendChild(opt);
                });
            }

            document.getElementById('stat-total').innerText = data.total;
            document.getElementById('stat-open').innerText = data.open;
            document.getElementById('stat-closed').innerText = data.closed;
            document.getElementById('stat-overdue').innerText = data.overdue;
            drawPieChart(data.open, data.closed, data.overdue);
            
            const summaryBody = document.getElementById('findingSummaryBody');
            let summaryHtml = '';
            const keys = ["nc1 (Major NC)", "nc2 (Minor NC) HMM", "nc2 (Minor NC) LPB", "OFI HMM", "OFI LPB", "Comment HMM", "Comment LPB"];
            
            keys.forEach(key => {
                const item = data.findingSummary[key];
                summaryHtml += `<tr><td class="text-start ps-3">${key}</td><td class="fw-bold">${item.count}</td><td>${item.car}</td><td>${item.ev}</td></tr>`;
            });
            summaryBody.innerHTML = summaryHtml;

            const tbody = document.getElementById('tableBody');
            let html = '';
            data.records.forEach(raw => {
                const r = {};
                for (let key in raw) r[key] = (raw[key] === undefined || raw[key] === null || raw[key] === "undefined" || raw[key] === "") ? "" : raw[key];

                let actionBtn = '';
                // 🌟 อัปเดตสิทธิ์: Admin, คนสร้าง (Auditor), และ คนรับผิดชอบ (Auditee) สามารถกด Edit ได้
                if (data.isAdmin || data.currentUserEmail === r.createdBy || data.currentUserEmail === r.carEmail) {
                    let recordJson = encodeURIComponent(JSON.stringify(raw));
                    actionBtn = `<button class="btn btn-sm btn-primary shadow-sm" onclick="editRecord('${recordJson}')"><i class="bi bi-pencil-square"></i> Edit</button>`;
                } else {
                    actionBtn = `<button class="btn btn-sm btn-light text-muted border" disabled title="Locked"><i class="bi bi-lock-fill"></i></button>`;
                }

                let statusClass = r.statusCar === 'Open' ? 'bg-warning text-dark' : 'bg-success';
                let carFileLink = (r.carFiles && r.carFiles.length > 10) ? `<br><a href="${r.carFiles}" target="_blank" class="badge bg-primary text-decoration-none mt-2"><i class="bi bi-paperclip"></i> View CAR File</a>` : '';
                let whyFileLink = (r.whyFiles && r.whyFiles.length > 10) ? `<br><a href="${r.whyFiles}" target="_blank" class="badge bg-primary text-decoration-none mt-2"><i class="bi bi-paperclip"></i> View 5 Why File</a>` : '';
                let yokoFileLink = (r.yokotenFiles && r.yokotenFiles.length > 10) ? `<br><a href="${r.yokotenFiles}" target="_blank" class="badge bg-primary text-decoration-none mt-2"><i class="bi bi-paperclip"></i> View Yoko-ten File</a>` : '';
                
                html += `<tr>
                    <td class="text-center">${actionBtn}</td> 
                    <td class="text-center">${formatEnglishDate(r.date)}</td>
                    <td class="text-center">${r.plant}</td>
                    <td class="small text-wrap" style="min-width: 150px;">${r.findings}</td>
                    <td class="text-center"><span class="badge bg-light text-dark border">${r.dept}</span></td>
                    <td class="small text-wrap" style="min-width: 250px;">${r.carDetail} ${carFileLink}</td>
                    <td class="small text-wrap" style="min-width: 250px;">${r.whyDetail} ${whyFileLink}</td>
                    <td class="small text-center">${r.carEmail}</td>
                    <td class="text-center">${formatEnglishDate(r.carDeadline)}</td>
                    <td class="text-center">${formatEnglishDate(r.carIntDeadline)}</td>
                    <td class="text-center"><span class="badge ${statusClass}">${r.statusCar}</span></td>
                    <td class="text-center">${formatEnglishDate(r.evDeadline)}</td>
                    <td class="small text-wrap" style="min-width: 150px;">${r.fu1}</td>
                    <td class="small text-wrap" style="min-width: 150px;">${r.fu2}</td>
                    <td class="small text-wrap" style="min-width: 150px;">${r.fu3}</td>
                    <td class="small text-wrap" style="min-width: 200px;">${r.yokotenDetail} ${yokoFileLink}</td>
                </tr>`;
            });
            tbody.innerHTML = html || '<tr><td colspan="16" class="text-center py-5 text-muted">No records found for this year.</td></tr>';
        }).getDashboardData(selectedYear);
    }
    function editRecord(jsonStr) {
        let r = JSON.parse(decodeURIComponent(jsonStr));
        
        const form = document.getElementById('formSection');
        form.style.display = 'block';
        form.scrollIntoView({ behavior: 'smooth' });

        document.getElementById('hiddenAuditId').value = r.id; 
        document.getElementById('btnSubmit').innerHTML = '<i class="bi bi-arrow-repeat"></i> UPDATE DATA'; 
        document.getElementById('btnSubmit').classList.replace('btn-mnb', 'btn-warning');

        const formatDateForInput = (dateStr) => {
            if(!dateStr || dateStr.trim() === "") return "";
            let d = new Date(dateStr);
            if (isNaN(d.getTime())) return "";
            return d.toISOString().split('T')[0];
        };

        document.getElementById('auditDate').value = formatDateForInput(r.date);
        document.getElementById('plant').value = r.plant;
        document.getElementById('dept').value = r.dept;
        document.getElementById('detail').value = r.detail;
        
        document.querySelectorAll('input[name="finding"]').forEach(cb => { cb.checked = false; });
        if(r.findings) {
            let fArr = r.findings.split(',').map(s => s.trim());
            document.querySelectorAll('input[name="finding"]').forEach(cb => {
                if(fArr.includes(cb.value)) cb.checked = true;
            });
        }

        document.getElementById('carDetail').value = r.carDetail;
        document.getElementById('whyDetail').value = r.whyDetail;
        document.getElementById('carEmail').value = r.carEmail;
        document.getElementById('carDeadline').value = formatDateForInput(r.carDeadline);
        document.getElementById('carIntDeadline').value = formatDateForInput(r.carIntDeadline);
        document.getElementById('evDeadline').value = formatDateForInput(r.evDeadline);
        document.getElementById('fu1').value = r.fu1;
        document.getElementById('fu2').value = r.fu2;
        document.getElementById('fu3').value = r.fu3;
        document.getElementById('yokotenDetail').value = r.yokotenDetail;

        document.getElementById('statusCar').value = r.statusCar || "Open";
        document.getElementById('evStatus').value = r.evStatus || "Open";

        if (currentIsAdmin) {
            document.getElementById('statusCar').disabled = false;
            document.getElementById('evStatus').disabled = false;
        } else {
            document.getElementById('statusCar').disabled = true;
            document.getElementById('evStatus').disabled = true;
        }
    }

    function drawPieChart(open, closed, overdue) {
        const ctx = document.getElementById('statusChart').getContext('2d');
        if(statusChartInstance) statusChartInstance.destroy();

        const total = open + closed + overdue;
        const slicePercentPlugin = {
            id: 'slicePercent',
            afterDraw: (chart) => {
                const ctx = chart.ctx;
                chart.data.datasets.forEach((dataset, i) => {
                    const meta = chart.getDatasetMeta(i);
                    meta.data.forEach((element, index) => {
                        const dataVal = dataset.data[index];
                        if (dataVal > 0) { 
                            const percent = Math.round((dataVal / total) * 100) + '%';
                            const position = element.tooltipPosition();
                            ctx.fillStyle = '#ffffff'; 
                            ctx.font = 'bold 13px sans-serif';
                            ctx.textAlign = 'center'; ctx.textBaseline = 'middle';
                            ctx.shadowColor = 'rgba(0, 0, 0, 0.7)'; ctx.shadowBlur = 4;
                            ctx.shadowOffsetX = 1; ctx.shadowOffsetY = 1;
                            ctx.fillText(percent, position.x, position.y);
                            ctx.shadowColor = 'transparent';
                        }
                    });
                });
            }
        };
        
        statusChartInstance = new Chart(ctx, {
            type: 'doughnut',
            data: {
                labels: ['Open', 'Closed', 'Overdue'],
                datasets: [{ data: [open, closed, overdue], backgroundColor: ['#f59e0b', '#10b981', '#ef4444'], borderWidth: 2, borderColor: '#ffffff' }]
            },
            options: {
                responsive: true, maintainAspectRatio: false,
                plugins: { 
                    legend: { display: false },
                    tooltip: {
                        callbacks: {
                            label: function(context) {
                                let label = context.label || '';
                                if (label) label += ': ';
                                if (context.parsed !== null) label += context.parsed + ' (' + Math.round((context.parsed / total) * 100) + '%)';
                                return label;
                            }
                        }
                    }
                },
                cutout: '55%' 
            },
            plugins: [slicePercentPlugin] 
        });
    }

    function filterTable() {
        const plantFilter = document.getElementById('plantFilter').value.toLowerCase();
        const findingCheckboxes = document.querySelectorAll('.finding-filter-cb:checked');
        const selectedFindings = Array.from(findingCheckboxes).map(cb => cb.value.toLowerCase());

        const tbody = document.getElementById('tableBody');
        const trs = tbody.getElementsByTagName('tr');
        for (let i = 0; i < trs.length; i++) {
            if (trs[i].cells.length <= 2) continue;
            let plantCell = trs[i].cells[2]; 
            let findingsCell = trs[i].cells[3]; 

            if (plantCell && findingsCell) {
                let plantText = plantCell.textContent.toLowerCase();
                let findingsText = findingsCell.textContent.toLowerCase();

                let matchPlant = (plantFilter === "" || plantText.includes(plantFilter));
                let matchFindings = true;
                if (selectedFindings.length > 0) matchFindings = selectedFindings.some(word => findingsText.includes(word));

                if (matchPlant && matchFindings) trs[i].style.display = '';
                else trs[i].style.display = 'none';
            }
        }
    }

    function clearFindingFilters() {
        document.querySelectorAll('.finding-filter-cb').forEach(cb => cb.checked = false);
        filterTable(); 
    }
</script>
