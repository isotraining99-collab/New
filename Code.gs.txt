function uploadFile(base64Data, fileName) {
  try {
    var folder = DriveApp.getFolderById(FOLDER_ID);
    var splitBase = base64Data.split(',');
    var type = splitBase[0].split(';')[0].replace('data:', '');
    var blob = Utilities.newBlob(Utilities.base64Decode(splitBase[1]), type, fileName);
    var file = folder.createFile(blob);
    file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
    return file.getUrl();
  } catch (e) { return ""; }
}

function saveAuditData(form) {
  try {
    var email = Session.getActiveUser().getEmail(); 
    var year = form.auditDate ? form.auditDate.substring(0, 4) : new Date().getFullYear(); 
    var ss = SpreadsheetApp.openById(SHEET_ID).getSheetByName("Database_TUV");

    if (form.hiddenAuditId) {
      var data = ss.getDataRange().getValues();
      var rowIndex = -1;
      for (var i = 1; i < data.length; i++) {
        if (data[i][1] === form.hiddenAuditId) { rowIndex = i + 1; break; }
      }
      
      if (rowIndex > -1) {
        var oldRow = data[rowIndex - 1];
        var rFile = form.reportFiles || oldRow[5];
        var cFile = form.carFiles || oldRow[9];
        var wFile = form.whyFiles || oldRow[11];
        var yFile = form.yokotenFiles || oldRow[22];

        var updateValues = [
          form.auditDate, form.plant, form.detail, rFile, form.findings, form.dept,
          form.carDetail, cFile, form.whyDetail, wFile, form.carEmail, form.carDeadline,
          form.carIntDeadline, form.statusCar, form.evDeadline, form.evStatus,
          form.fu1, form.fu2, form.fu3, form.yokotenDetail, yFile, year
        ];
        ss.getRange(rowIndex, 3, 1, 22).setValues([updateValues]);
        return "✅ Update ID: " + form.hiddenAuditId + " success!";
      }
    } else {
      var newAuditId = "TUV-" + Utilities.formatDate(new Date(), "GMT+7", "yyMMdd-HHmmss");
      ss.appendRow([
        new Date(), newAuditId, form.auditDate, form.plant, form.detail, 
        form.reportFiles, form.findings, form.dept, form.carDetail, 
        form.carFiles, form.whyDetail, form.whyFiles, form.carEmail, 
        form.carDeadline, form.carIntDeadline, form.statusCar, 
        form.evDeadline, form.evStatus, form.fu1, form.fu2, form.fu3, 
        form.yokotenDetail, form.yokotenFiles, year, email 
      ]);
      return "✅ Save ID: " + newAuditId + " success!";
    }
  } catch (e) { throw new Error(e.message); }
}

function getDashboardData(selectedYear) {
  var email = Session.getActiveUser().getEmail();
  var ss = SpreadsheetApp.openById(SHEET_ID);
  var sheet = ss.getSheetByName("Database_TUV");
  var adminSheet = ss.getSheetByName("AdminList");
  
  var isAdmin = false;
  if (adminSheet) {
    var admins = adminSheet.getRange("A2:A").getValues().flat().filter(String);
    isAdmin = admins.includes(email);
  }

  var data = sheet.getDataRange().getDisplayValues();
  var stats = { 
    currentUserEmail: email, isAdmin: isAdmin, availableYears: [],
    total: 0, open: 0, closed: 0, overdue: 0, records: [],
    findingSummary: {
      "nc1 (Major NC)": { count: 0, car: "-", ev: "-" }, "nc2 (Minor NC) HMM": { count: 0, car: "-", ev: "-" },
      "nc2 (Minor NC) LPB": { count: 0, car: "-", ev: "-" }, "OFI HMM": { count: 0, car: "-", ev: "-" },
      "OFI LPB": { count: 0, car: "-", ev: "-" }, "Comment HMM": { count: 0, car: "-", ev: "-" },
      "Comment LPB": { count: 0, car: "-", ev: "-" }
    }
  };
  
  if (data.length <= 1) return stats;
  
  var yearsSet = new Set();
  for (var i = 1; i < data.length; i++) { if (data[i][23]) yearsSet.add(data[i][23]); }
  stats.availableYears = Array.from(yearsSet).sort().reverse();

  var today = new Date();
  today.setHours(0,0,0,0);
  
  for (var i = data.length - 1; i >= 1; i--) { 
    var row = data[i];
    if (!row || row.length === 0) continue; 
    
    var rowYear = row[23] || "";
    if (selectedYear && selectedYear !== "All" && rowYear !== selectedYear) continue;

    var status = row[15] || "";
    var deadlineStr = row[13] || "";
    var findingsStr = row[6] || ""; 
    
    for (var key in stats.findingSummary) {
      if (findingsStr.indexOf(key) !== -1) {
        stats.findingSummary[key].count++;
        if (stats.findingSummary[key].car === "-") {
          stats.findingSummary[key].car = row[13] || "-";
          stats.findingSummary[key].ev = row[16] || "-";
        }
      }
    }
    
    var findingItems = findingsStr ? findingsStr.split(',').length : 0;
    stats.total += findingItems;
    
    if (status === "Open") {
      stats.open++;
      if (deadlineStr) {
        var deadlineDate = new Date(deadlineStr);
        if(deadlineDate < today) stats.overdue++;
      }
    } else if (status === "Close") {
      stats.closed++;
    }
    
    stats.records.push({
      id: row[1] || "", 
      date: row[2] || "", plant: row[3] || "", detail: row[4] || "", reportFiles: row[5] || "",
      findings: row[6] || "", dept: row[7] || "", carDetail: row[8] || "", carFiles: row[9] || "",
      whyDetail: row[10] || "", whyFiles: row[11] || "", c: row[12] || "",
      carDeadline: row[13] || "", carIntDeadline: row[14] || "", statusCar: row[15] || "", 
      evDeadline: row[16] || "", evStatus: row[17] || "", fu1: row[18] || "", fu2: row[19] || "", 
      fu3: row[20] || "", yokotenDetail: row[21] || "", yokotenFiles: row[22] || "",
      createdBy: row[24] || "" 
    });
  }
  return stats;
}

// 🌟 ฟังก์ชันดึงข้อมูล Shared Report แยกตามปี
function getSharedReport(year) {
  var ss = SpreadsheetApp.openById(SHEET_ID);
  var sheet = ss.getSheetByName("ReportData");
  if (!sheet) return ""; 

  // ถ้าไม่ได้เลือกปี หรือเลือก All ให้ดึงของปีปัจจุบัน
  //if (!year || year === "All") year = new Date().getFullYear().toString();
  
  var data = sheet.getDataRange().getValues();
  for (var i = 0; i < data.length; i++) {
    if (data[i][0].toString() === year.toString()) {
      return data[i][1]; // คืนค่าเนื้อหา HTML ของปีนั้น
    }
  }
  return ""; // ถ้ายังไม่มีข้อมูลของปีนั้น ให้คืนค่าช่องว่าง
}

// 🌟 ฟังก์ชันบันทึก Shared Report แยกตามปี
function saveSharedReport(htmlContent, year) {
  var ss = SpreadsheetApp.openById(SHEET_ID);
  var sheet = ss.getSheetByName("ReportData");
  
  if (!sheet) {
    sheet = ss.insertSheet("ReportData");
  }

  if (!year || year === "All") year = new Date().getFullYear().toString();
  
  var data = sheet.getDataRange().getValues();
  var found = false;
  
  // วนลูปหาว่าปีนี้มีบันทึกไว้หรือยัง
  for (var i = 0; i < data.length; i++) {
    if (data[i][0].toString() === year.toString()) {
      sheet.getRange(i + 1, 2).setValue(htmlContent); // ถ้าเจอ อัปเดตทับ
      found = true;
      break;
    }
  }
  
  // ถ้าหาไม่เจอ แปลว่าเป็นปีใหม่ ให้เพิ่มแถวใหม่
  if (!found) {
    sheet.appendRow([year, htmlContent]);
  }
  return true;
}
